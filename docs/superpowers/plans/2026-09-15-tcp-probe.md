```csharp
// Core/ConCore/ProbeSpec.cs
using System.Net;

namespace ConCore;

public enum ProbeKind { Icmp, Tcp }

/// <summary>
/// How a target is probed: ICMP echo (the default) or a TCP connect where
/// any stack response — SYN-ACK or RST — counts as a reply. Stored in
/// configuration as one optional string ("tcp:80"); absent means ICMP, so
/// every pre-existing config, snapshot, and seed is valid unchanged.
/// </summary>
public readonly record struct ProbeSpec(ProbeKind Kind, int Port = 0)
{
    public static ProbeSpec Icmp => default;

    public static bool TryParse(string? text, out ProbeSpec spec)
    {
        spec = Icmp;
        if (string.IsNullOrWhiteSpace(text))
            return true;
        var parts = text.Trim().ToLowerInvariant().Split(':');
        if (parts is ["icmp"])
            return true;
        if (parts is ["tcp", var portText] &&
            int.TryParse(portText, out var port) && port is >= 1 and <= 65535)
        {
            spec = new ProbeSpec(ProbeKind.Tcp, port);
            return true;
        }
        return false;
    }

    public override string ToString() => Kind == ProbeKind.Tcp ? $"tcp:{Port}" : "icmp";

    /// <summary>The storage form: null for the ICMP default.</summary>
    public string? ToConfigString() => Kind == ProbeKind.Icmp ? null : ToString();
}

/// <summary>
/// Politeness policy: on your own network probe anything; on the public
/// internet only ports where a bare connect-and-close is indistinguishable
/// from normal client traffic. Hostnames count as public.
/// </summary>
public static class ProbePolicy
{
    public static readonly int[] PublicPorts = [80, 443, 53];

    public static bool IsAllowed(string address, ProbeSpec probe, out string? reason)
    {
        reason = null;
        if (probe.Kind != ProbeKind.Tcp || IsPrivate(address) || PublicPorts.Contains(probe.Port))
            return true;
        reason = $"TCP probes to public addresses are limited to ports " +
                 $"{string.Join(", ", PublicPorts)} — '{address}' is not a private address.";
        return false;
    }

    public static bool IsPrivate(string address)
    {
        if (!IPAddress.TryParse(address, out var ip))
            return false; // hostname: treat as public
        if (IPAddress.IsLoopback(ip))
            return true;
        var b = ip.GetAddressBytes();
        if (b.Length != 4)
            return false; // IPv6 unsupported end to end
        return b[0] == 10
            || (b[0] == 172 && b[1] is >= 16 and <= 31)
            || (b[0] == 192 && b[1] == 168)
            || (b[0] == 169 && b[1] == 254);
    }
}
```

```csharp
// ConInfrastructure/TcpConnectTransceiver.cs
using System.Diagnostics;
using System.Net;
using System.Net.Sockets;
using ConCore;

namespace ConInfrastructure;

/// <summary>
/// TCP-connect probe: one fresh socket per probe, no data ever sent, closed
/// cleanly (FIN) when accepted. Any stack response counts as a reply —
/// SYN-ACK ("Accepted") or RST ("Refused") — because the question is "can I
/// reach this device", not "is the service healthy". Only silence is loss.
/// The timeout races ConnectAsync because the OS SYN-retry schedule ignores
/// caller deadlines.
/// </summary>
public sealed class TcpConnectTransceiver(IPAddress ipAddress, int port,
    CancellationToken cancellationToken) : IPingTransceiver
{
    public IPAddress IpAddress => ipAddress;

    public async Task<PingResult> PingAsync(int timeoutMs)
    {
        var stopwatch = Stopwatch.StartNew();
        using var socket = new Socket(ipAddress.AddressFamily, SocketType.Stream, ProtocolType.Tcp);
        using var timeout = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        timeout.CancelAfter(timeoutMs);
        try
        {
            await socket.ConnectAsync(new IPEndPoint(ipAddress, port), timeout.Token).ConfigureAwait(false);
            return new PingResult(ipAddress, true, stopwatch.ElapsedMilliseconds, "Accepted");
        }
        catch (SocketException ex) when (ex.SocketErrorCode == SocketError.ConnectionRefused)
        {
            return new PingResult(ipAddress, true, stopwatch.ElapsedMilliseconds, "Refused");
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
            return new PingResult(ipAddress, false, 0, "Cancelled");
        }
        catch (OperationCanceledException)
        {
            return new PingResult(ipAddress, false, 0, "TimedOut");
        }
        catch (SocketException ex)
        {
            return new PingResult(ipAddress, false, 0, ex.SocketErrorCode.ToString());
        }
    }

    public void Dispose() { }
}
```

```csharp
// Programs/ConWatch/Api/ProbeTestEndpoint.cs
using System.Net;
using ConCore;

namespace ConWatch.Api;

/// <summary>One-click probe discovery for the Targets form: the SERVER
/// (where monitoring runs) tries ICMP plus the policy-allowed TCP ports
/// once each, so the form can preselect a probe that actually answers.</summary>
public static class ProbeTestEndpoint
{
    public record Request(string Address, int? Port = null);
    private const int TimeoutMs = 2000;

    public static void MapProbeTest(this RouteGroupBuilder locked) =>
        locked.MapPost("/targets/probe-test", async (Request request,
            IPingTransceiverFactory factory, IDomainNameResolverFactory resolvers,
            CancellationToken ct) =>
        {
            IPAddress? ip = null;
            if (!IPAddress.TryParse(request.Address, out ip))
            {
                using var resolver = resolvers.Create(ct);
                var resolution = await resolver.ResolveAsync(request.Address, 5000);
                ip = resolution.IPv4Addresses.FirstOrDefault();
            }
            if (ip is null)
                return Results.Ok(new { request.Address, ResolvedIp = (string?)null,
                    Results = Array.Empty<object>() });

            var ports = (ProbePolicy.IsPrivate(request.Address)
                    ? new[] { 80, 443, 8080, 8443 } : ProbePolicy.PublicPorts)
                .Concat(request.Port is { } p &&
                    ProbePolicy.IsAllowed(request.Address, new ProbeSpec(ProbeKind.Tcp, p), out _)
                    ? [p] : Array.Empty<int>())
                .Distinct().ToList();
            var candidates = new List<ProbeSpec> { ProbeSpec.Icmp };
            candidates.AddRange(ports.Select(port => new ProbeSpec(ProbeKind.Tcp, port)));

            var results = await Task.WhenAll(candidates.Select(async probe =>
            {
                using var transceiver = factory.Create(ip, probe, ct);
                var result = await transceiver.PingAsync(TimeoutMs);
                return new { Probe = probe.ToString(), Reply = result.Success,
                    RttMs = result.RoundtripTime, Message = result.Message };
            }));
            return Results.Ok(new { request.Address, ResolvedIp = ip.ToString(), Results = results });
        });
}
```

```csharp
// Programs/ConWatch/RuntimeHealth.cs
using ConEngine;
using ConPersistence;

namespace ConWatch;

/// <summary>Warnings derivable only from the live engine (ConfigHealth is
/// config-only and cannot see runtime counters).</summary>
public static class RuntimeHealth
{
    public static List<HealthWarning> RefusalWarnings(WatcherEngine? engine)
    {
        var warnings = new List<HealthWarning>();
        foreach (var monitor in engine?.Monitors.Values ?? [])
        {
            if (!monitor.PersistentlyRefused)
                continue;
            warnings.Add(new HealthWarning("tcp-refused",
                $"{monitor.Target.Name} has refused every connection on " +
                $"{monitor.Target.Probe} ({monitor.RefusedReplies} probes) — nothing is " +
                "listening there. The device is up, but use the Test button to find a port " +
                "that accepts."));
        }
        return warnings;
    }
}
```
